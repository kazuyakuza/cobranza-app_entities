### 5.2 Angular Reactive Form Example (`docs/usage-angular.md`)

```typescript
import { Component } from '@angular/core';
import { FormBuilder, FormGroup, Validators } from '@angular/forms';
import { Client } from '@cobranza-apps/entities';

type ClientFormPayload = Omit<Client, 'id' | 'createdAt' | 'updatedAt' | 'updatedBy'>;

@Component({ selector: 'app-client-form', template: '...' })
export class ClientFormComponent {
  form: FormGroup = this.fb.group({
    fullName: ['', Validators.required],
    email: ['', Validators.email],
    phone: [''],
    taxId: [''],
    companyId: ['', Validators.required],
    active: [true],
  });

  constructor(private fb: FormBuilder) {}

  getPayload(): ClientFormPayload {
    return this.form.value;
  }
}
```

### 5.3 OpenAPI Decorator Example (`docs/openapi-examples.md`)

```typescript
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import { Debt as DebtInterface, DebtStatus } from '@cobranza-apps/entities';

export class Debt implements DebtInterface {
  @ApiProperty({ format: 'uuid', description: 'Primary key' })
  id: string;

  @ApiProperty({ format: 'uuid' })
  companyId: string;

  @ApiProperty()
  description: string;

  @ApiProperty({ type: String, description: 'Original amount as decimal string' })
  totalAmount: string;

  @ApiProperty({ enum: DebtStatus, enumName: 'DebtStatus' })
  status: DebtStatus;
}
```
